1 架构与页面路由 基础

本平台为纯前端 SPA 风格多页面应用，页面间通过 window.location.href 跳转，登录态持久化在 localStorage。企业端与 OPC 端工作台权限隔离，未认证用户功能受限。

1.2 登录态校验逻辑

// 每个受保护页面在 <script> 顶部执行
var user = JSON.parse(localStorage.getItem("opc_user") || "null");
if (!user) {
  window.location.href = "login.html";
}
// 登录态过期：7 天 = 7 * 24 * 60 * 60 * 1000 ms
var EXPIRE_MS = 604800000;
if (Date.now() - user.time > EXPIRE_MS) {
  localStorage.removeItem("opc_user");   // 先 removeItem 再跳转
  window.location.href = "login.html";
}
⚠ 开发者注意 user.time 是登录时写入的 Date.now() 毫秒时间戳。过期后必须 先 removeItem 再跳转，否则用户回退时仍能访问受保护页面。

4.5 认证状态实时同步

// 定时同步
setInterval(syncAuthStatus, 3000);

// 焦点同步
window.addEventListener("focus", syncAuthStatus);

function syncAuthStatus() {
  var user = JSON.parse(localStorage.getItem("opc_user"));
  var accounts = JSON.parse(localStorage.getItem("opc_accounts") || "[]");
  var acc = accounts.find(a => a.accountId === user.accountId);
  if (acc) {
    user.authStatus = acc.authStatus;
    user.authRole = acc.authRole;
    user.avatar = acc.avatar;
    localStorage.setItem("opc_user", JSON.stringify(user));
    updateBadgeAndAvatar();  // 更新徽章和头像
  }
}

6.5 文件保存流程

function saveFile(file, fileType, uploadUser) {
  // 1. 校验大小
  var maxMB = (fileType === "avatar") ? 2 : 10;
  if (file.size > maxMB * 1024 * 1024) {
    Toast.error("文件大小不能超过" + maxMB + "MB");
    return null;
  }
  // 2. 校验类型
  var allowed = ["image/jpeg", "image/png", "image/gif", "image/webp", "image/bmp", "application/pdf"];
  if (!allowed.includes(file.type)) {
    Toast.error("不支持的文件类型");
    return null;
  }
  // 3. 生成记录并保存
  var files = JSON.parse(localStorage.getItem("opc_files") || "[]");
  var fileId = files.length + 1;
  var record = {
    fileId, fileName: file.name, filePath: generatePath(fileId, file.name),
    fileSize: file.size, contentType: file.type, uploadUser,
    createTime: Date.now(), fileType: fileType || ""
  };
  files.push(record);
  localStorage.setItem("opc_files", JSON.stringify(files));
  return record;
}

function addLog(user, type, field, desc) {
  var logs = JSON.parse(localStorage.getItem("opc_change_logs") || "[]");
  logs.push({
    time: Date.now(),
    userId: user.accountId,
    userName: user.username,
    type: type,        // avatar | profile | auth | status | file
    field: field,
    desc: desc
  });
  localStorage.setItem("opc_change_logs", JSON.stringify(logs));
}
